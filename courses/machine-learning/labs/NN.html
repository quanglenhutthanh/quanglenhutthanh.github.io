Ghi chú

Phân loại chữ số viết tay MNIST bằng Keras, minh họa 3 cách định nghĩa model:

  • Sequential API — xếp chồng layer tuần tự (Flatten → Dense → BatchNorm → … → softmax).
  • Functional API — dùng Input/gọi layer như hàm, linh hoạt hơn cho graph phức tạp (multi-input/output).
  • Subclassing tf.keras.Model — tự định nghĩa __init__/call; model chỉ “build” (hiện số param) sau khi được gọi lần đầu với dữ liệu thật.

Ngoài ra còn có: - ExponentialDecay learning rate schedule + callback EarlyStopping/TensorBoard. - Custom training loop thủ công bằng tf.GradientTape (thay vì model.fit), tự tính gradient và cập nhật qua optimizer.apply_gradients.

Define NN using function

def create_model(shape=(28,28)):
  # Define the input layer
  inputs = tf.keras.Input(shape=shape, name='input_layer')
  # Flatten the input
  x = tf.keras.layers.Flatten(name='flatten_layer')(inputs)

  # Add the dense layers
  x = tf.keras.layers.Dense(256, activation='relu', name='dense_1')(x)
  x = tf.keras.layers.Dense(128, activation='relu', name='dense_2')(x)

  # Output layer
  outputs = tf.keras.layers.Dense(10, activation='softmax', name='output_layer')(x)

  # Create the model using the Functional API
  model_1 = tf.keras.Model(inputs=inputs, outputs=outputs, name='functional_model')
  return model_1

import tensorflow as tf

class MyModel(tf.keras.Model):
  def __init__(self, num_classes=10):
    super(MyModel, self).__init__()
    self.flatten = tf.keras.layers.Flatten(name='flatten_layer')
    self.dense1 = tf.keras.layers.Dense(256, activation='relu', name='dense_1')
    self.dense2 = tf.keras.layers.Dense(128, activation='relu', name='dense_2')
    self.output_layer = tf.keras.layers.Dense(num_classes, activation='softmax', name='output_layer')

  def call(self, inputs):
    x = self.flatten(inputs)
    x = self.dense1(x)
    x = self.dense2(x)
    return self.output_layer(x)

# Create an instance of the subclassed model

import numpy as np

# Normalize the input data
x_train_normalized = x_train / 255.0
x_test_normalized = x_test / 255.0

# Define loss and optimizer
loss_object = tf.keras.losses.SparseCategoricalCrossentropy()
optimizer = tf.keras.optimizers.Adam(learning_rate=0.001)

@tf.function
def train_step(images, labels, train_loss_metric, train_accuracy_metric):
  with tf.GradientTape() as tape:
    predictions = model_1(images, training=True)
    loss = loss_object(labels, predictions)
  gradients = tape.gradient(loss, model_1.trainable_variables)
  optimizer.apply_gradients(zip(gradients, model_1.trainable_variables))

  train_loss_metric(loss)
  train_accuracy_metric(labels, predictions)

@tf.function
def val_step(images, labels, val_loss_metric, val_accuracy_metric):
  predictions = model_1(images, training=False)
  t_loss = loss_object(labels, predictions)

  val_loss_metric(t_loss)
  val_accuracy_metric(labels, predictions)

# Create TensorFlow Datasets for batching
BATCH_SIZE = 128
train_dataset = tf.data.Dataset.from_tensor_slices((x_train_normalized, y_train)).shuffle(10000).batch(BATCH_SIZE)
test_dataset = tf.data.Dataset.from_tensor_slices((x_test_normalized, y_test)).batch(BATCH_SIZE)

EPOCHS = 5

for epoch in range(EPOCHS):
  # Re-instantiate the metrics at the start of each epoch to effectively reset them
  train_loss = tf.keras.metrics.Mean(name='train_loss')
  train_accuracy = tf.keras.metrics.SparseCategoricalAccuracy(name='train_accuracy')
  val_loss = tf.keras.metrics.Mean(name='val_loss')
  val_accuracy = tf.keras.metrics.SparseCategoricalAccuracy(name='val_accuracy')

  for images, labels in train_dataset:
    train_step(images, labels, train_loss, train_accuracy)

  for val_images, val_labels in test_dataset:
    val_step(val_images, val_labels, val_loss, val_accuracy)

  print(f"Epoch {epoch + 1}, "
        f"Loss: {train_loss.result():.4f}, "
        f"Accuracy: {train_accuracy.result():.4f}, "
        f"Val Loss: {val_loss.result():.4f}, "
        f"Val Accuracy: {val_accuracy.result():.4f}")
